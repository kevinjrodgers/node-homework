# Task 1 Find the total price of each of the first 5 orders, ordered by order_id
SELECT o.order_id, SUM(p.price * l.quantity) AS total_price FROM products AS p INNER JOIN line_items AS l ON p.product_id = l.product_id INNER JOIN orders AS o ON l.order_id = o.order_id GROUP BY o.order_id ORDER BY o.order_id ASC LIMIT 5;
