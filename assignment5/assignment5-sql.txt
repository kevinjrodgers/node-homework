# Task 1: Find the total price of each of the first 5 orders, ordered by order_id
SELECT o.order_id, SUM(p.price * l.quantity) AS total_price FROM products AS p JOIN line_items AS l ON p.product_id = l.product_id INNER JOIN orders AS o ON l.order_id = o.order_id GROUP BY o.order_id ORDER BY o.order_id ASC LIMIT 5;


# Task 2: Understanding Subqueries
SELECT c.customer_name, AVG(total_price) AS average_order_price FROM customers AS c JOIN (SELECT o.customer_id, o.order_id, SUM(p.price * l.quantity) AS total_price FROM products AS p JOIN line_items AS l ON p.product_id = l.product_id JOIN orders AS o ON l.order_id = o.order_id GROUP BY o.customer_id, o.order_id) AS t ON c.customer_id = t.customer_id GROUP BY c.customer_id, c.customer_name ORDER BY c.customer_name;

# Task 4: Aggregation with Having
SELECT first_name, last_name, COUNT(order_id) AS order_count FROM employees AS e JOIN orders AS o ON e.employee_id = o.employee_id GROUP BY e.employee_id, first_name, last_name HAVING COUNT(order_id) > 5 ORDER BY last_name;
